\chapter{Notation}
\section{Special operators and symbols}
\subsection*{Identity matrix}
\begin{equation}
  \T{I}_{n \times n}
  =
  \begin{bmatrix}
    1 & 0 & \cdots & 0 \\
    0 & 1 & \cdots & 0 \\
    \vdots & \vdots & \ddots & \vdots \\
    0 & 0 & \cdots & 1
  \end{bmatrix}
  \label{eq:identity_matrix}
\end{equation}

\subsection*{Skew-symmetric matrix}
\begin{equation}
  [\mathbf{v}]_\times =
  \begin{bmatrix}
    0 & -v_z & v_y \\
    v_z & 0 & -v_x \\
    -v_y & v_x & 0
  \end{bmatrix}
  \label{eq:skew_symmetric_matrix}
\end{equation}


\subsection*{Inner product}
\begin{equation}
  \iprd{\T{u}}{\T{v}} = \T{u}^{\mathsf{T}} \T{v}
  \label{eq:inner_product}
\end{equation}

\subsection*{Cross product}
\begin{equation}
  \mathbf{a} \times \mathbf{b} = [\mathbf{a}]_\times \mathbf{b} = 
  \begin{bmatrix}
      a_2 b_3 - a_3 b_2 \\
      a_3 b_1 - a_1 b_3 \\
      a_1 b_2 - a_2 b_1
  \end{bmatrix}
  \label{eq:cross_product}
\end{equation}

\subsubsection*{Mathematical spaces and sets}

$\mathbb{R}^n$ n-dimensional real space

$\SO{3}$ special orthogonal group of three-dimensional rotations

$\SE{3}$ special Euclidean group of three-dimensional rigid-body motions

$\mathbb{H}$ Hamilton's quaternion algebra

$S^3$ unit three-sphere representing unit quaternions

$\mathcal{Q}$ manipulator joint configuration space

$\mathcal{H}_d$ $d$-dimensional homotopy-parameter space

$\mathcal{U}$ uniform probability distribution

\subsubsection*{Operators and common matrices}

$\T{I}$ identity matrix

$\T{0}$ zero vector or zero matrix

$[\T{v}]_\times$ skew-symmetric matrix associated with vector $\T{v}$

\subsubsection*{Frames, rotations, and transformations}

$\T{R}$ rotation matrix

$\T{R}_{AB}$ rotation matrix that maps coordinates from frame $B$ to frame $A$

$\T{p}_{AB}$ position of the origin of frame $B$ expressed in frame $A$

$\T{T}$ homogeneous transformation matrix

$\T{T}_{AB}$ homogeneous transformation from frame $B$ to frame $A$

\subsubsection*{Quaternions and attitude representations}

$\T{q}$ configuration/state vector; in the spacecraft model, a unit quaternion, and in the manipulator model, a joint-position vector

In the spacecraft model, $\T{q} = [q_x, q_y, q_z, q_w]^{\mathsf{T}}$ satisfies
$q_x^2 + q_y^2 + q_z^2 + q_w^2 = 1$

$q_s$ scalar part of a quaternion

$\T{q}_v$ vector part of a quaternion

$\overline{\T{q}}$ quaternion conjugate

$\T{C}_q(\T{q})$ quaternion differential matrix

$\boldsymbol{\sigma}$ modified Rodrigues parameter vector

\subsubsection*{Trajectories and B-splines}

$\T{q}(t)$ time-dependent configuration/state trajectory; in the spacecraft model, a unit quaternion trajectory, and in the manipulator model, a joint-position trajectory

$\T{x}(t)$ generic state or trajectory vector; in IPSO, a particle-position vector

$\T{c}_i$ B-spline control vertex

$\T{v}_i$ derivative control vertex of a B-spline

$\boldsymbol{\kappa}$ B-spline knot vector

$\boldsymbol{\varphi}(s)$ B-spline basis vector

$\boldsymbol{\Phi}$ B-spline basis matrix evaluated at all sampling or via points

$\T{P}$ B-spline control-vertex matrix

$\T{Q}$ matrix of trajectory values evaluated at all sampling or via points

$\T{B}_0$, $\T{B}_f$ initial and final boundary basis matrices mapping exterior control vertices to boundary derivatives

$\T{Y}_0$, $\T{Y}_f$ initial and final boundary-condition matrices

$B_{\max}$, $B_{\min}$ upper and lower bounds for a control vertex

\subsubsection*{Manipulator dynamics and kinematics}

$\T{M}(\T{q})$ manipulator mass matrix

$\T{J}_{\mathrm{m}}$ diagonal matrix of joint motor inertias

$\T{J}_{\mathrm{load}}$ payload inertia matrix

$\T{J}_{g,e}$ geometric end-effector Jacobian

$\T{h}(\T{q},\dot{\T{q}})$ centripetal-Coriolis vector

$\T{g}(\T{q})$ gravity vector

$\boldsymbol{\tau}$ control torque vector

\subsubsection*{IPSO and nonlinear optimization}

$\T{w}$ nonlinear-programming decision vector

$\T{x}_{\min}$, $\T{x}_{\max}$ lower and upper IPSO search-space bounds

$p_{\max}$ maximum scalar MRP search bound

$f_j$ fitness of particle $j$

$v_{j,d}^{k}$ velocity of particle $j$ in dimension $d$ at iteration $k$

$p_{j,d}^{k}$, $l_{j,d}^{k}$, $g_d^{k}$ personal-best, local-best, and global-best particle positions

$w$, $c_p$, $c_l$, $c_g$ inertia, personal, local, and global weight coefficients

$\boldsymbol{\omega}$ angular velocity vector

$H$ Hamiltonian in the optimal-control formulation

\subsubsection*{Craig modified Denavit--Hartenberg parameters}

$a_{i-1}$ link length

$\alpha_{i-1}$ link twist

$d_{i-1}$ link offset

$\theta_{i-1}$ joint angle

In the Craig modified Denavit--Hartenberg convention, the transform from frame
$i$ to frame $i-1$ is
\begin{equation}
  {}^{i-1}\T{T}_{i} =
  \mathrm{Rot}_{x}(\alpha_{i-1})\,
  \mathrm{Trans}_{x}(a_{i-1})\,
  \mathrm{Rot}_{z}(\theta_i)\,
  \mathrm{Trans}_{z}(d_i).
\end{equation}
