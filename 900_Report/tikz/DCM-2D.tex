\begin{tikzpicture}[>=stealth, line width=1pt, scale=2.5]
    \def\alphaAngle{25}
    \def\axisLen{1.2}

    % Inertial frame
    \draw[->] (0,0) -- (\axisLen,0) node[right] {$\hat{\mathbf{B}}_x$};
    \draw[->] (0,0) -- (0,\axisLen) node[above] {$\hat{\mathbf{B}}_y$};

    % Body frame rotated by alpha Angle
    \draw[->] (0,0) -- (\alphaAngle:\axisLen) node[right] {$\hat{\mathbf{I}}_x$};
    \draw[->] (0,0) -- (90+\alphaAngle:\axisLen) node[above left] {$\hat{\mathbf{I}}_y$};

    % Common z-axis, pointing out of the page
    \filldraw[fill=white, draw=black] (0,0) circle (0.08);
    \filldraw[black] (0,0) circle (0.02);
    \node[anchor=north east] at (-0.05,-0.05) {$\hat{\mathbf{B}}_z = \hat{\mathbf{I}}_z$};

    % Rotation angle
    \draw[->, thin] (0.4,0) arc (0:\alphaAngle:0.4);
    \node at ({\alphaAngle/2}:0.5) {$\alpha$};
\end{tikzpicture}
